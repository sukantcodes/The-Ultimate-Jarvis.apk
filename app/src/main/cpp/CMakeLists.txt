cmake_minimum_required(VERSION 3.31.6)
project(ai-chat VERSION 1.0.0 LANGUAGES C CXX)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
include(FetchContent)
# Keep the upstream revision pinned. CI prefetches this archive with retries and
# supplies LLAMA_CPP_SRC, avoiding fragile long downloads inside CMake.
FetchContent_Declare(llama_cpp
 URL https://codeload.github.com/ggml-org/llama.cpp/tar.gz/da2ee1e7a640445fc562fe5529598cee1917e0f5
 DOWNLOAD_EXTRACT_TIMESTAMP TRUE
)
if(DEFINED ENV{LLAMA_CPP_SRC} AND IS_DIRECTORY "$ENV{LLAMA_CPP_SRC}")
 set(FETCHCONTENT_SOURCE_DIR_LLAMA_CPP "$ENV{LLAMA_CPP_SRC}" CACHE PATH "Prefetched llama.cpp sources" FORCE)
 message(STATUS "Using prefetched llama.cpp from $ENV{LLAMA_CPP_SRC}")
endif()
set(LLAMA_BUILD_COMMON ON CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(LLAMA_OPENSSL OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_BACKEND_DL OFF CACHE BOOL "" FORCE)
set(GGML_CPU_ALL_VARIANTS OFF CACHE BOOL "" FORCE)
set(GGML_LLAMAFILE OFF CACHE BOOL "" FORCE)
if(ANDROID_ABI STREQUAL "arm64-v8a")
 set(GGML_SYSTEM_ARCH "ARM")
 set(GGML_CPU_KLEIDIAI ON)
 set(GGML_OPENMP OFF)
elseif(ANDROID_ABI STREQUAL "x86_64")
 set(GGML_SYSTEM_ARCH "x86")
 set(GGML_CPU_KLEIDIAI OFF)
 set(GGML_OPENMP OFF)
else()
 message(FATAL_ERROR "Unsupported Android ABI: ${ANDROID_ABI}")
endif()
# CI prefetches KleidiAI with retries because GitHub release assets sometimes answer 504.
# When KLEIDIAI_SRC points at an extracted copy, llama.cpp/ggml uses it instead of downloading.
if(DEFINED ENV{KLEIDIAI_SRC} AND IS_DIRECTORY "$ENV{KLEIDIAI_SRC}")
 set(FETCHCONTENT_SOURCE_DIR_KLEIDIAI "$ENV{KLEIDIAI_SRC}" CACHE PATH "Prefetched KleidiAI sources" FORCE)
 message(STATUS "Using prefetched KleidiAI from $ENV{KLEIDIAI_SRC}")
endif()
FetchContent_MakeAvailable(llama_cpp)
set(LLAMA_SRC ${llama_cpp_SOURCE_DIR})
add_library(ai-chat SHARED ai_chat.cpp)
target_compile_definitions(ai-chat PRIVATE GGML_SYSTEM_ARCH=${GGML_SYSTEM_ARCH} GGML_CPU_KLEIDIAI=$<BOOL:${GGML_CPU_KLEIDIAI}> GGML_OPENMP=$<BOOL:${GGML_OPENMP}>)
target_include_directories(ai-chat PRIVATE ${LLAMA_SRC} ${LLAMA_SRC}/common ${LLAMA_SRC}/include ${LLAMA_SRC}/ggml/include ${LLAMA_SRC}/ggml/src)
target_link_libraries(ai-chat llama llama-common android log)
